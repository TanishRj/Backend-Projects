
1. Now we will write a new service in new file named `url.service.js` under `/services` directory which will insert the url with short-code provided
```js
// Importing db from index file of db

import {db} from '../db/index.js'

// Importing url table from url model

import {urlsTable} from '../models/index.js'

  

export async function newUrlInsert(shortCode, url, userId) {

    const [result] = await db.insert(urlsTable).values({

        // Getting short code

        shortCode,

        // targetUrl

        targetURL: url,

        // user id

        userId

    }).returning({

        id: urlsTable.id,

        shortCode: urlsTable.shortCode,

        targetURL: urlsTable.targetURL

    })

  

    // Returning result

    return result

}
```

2. Refactoring code in our `url.routes.js`
```js
// Importing url insertion method from url service file
import {newUrlInsert} from '../services/url.service.js'

// Getting userId
const userId = req.user.id

// Storing result from our method newUrlInsert
const result = await newUrlInsert(shortCode, url, userId)

// Returning result
return res.status(201).json(result)
```
