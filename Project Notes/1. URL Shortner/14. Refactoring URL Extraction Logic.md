
1. We will move the if user is logged in logic in the new middleware named `ensureAuthenticated` in file `auth.middleware.js`
```js
// Creating JS Doc for req, res and next

/**

 *

 * @param {import("express").Request} req

 * @param {import("express").Response} res

 * @param {import("express").NextFunction} next

 */

  

// Creating a middleware to check user is always authenticated

export function ensureAuthenticated(req, res, next) {

    if (!req.user || !req.user.id) {

        return res

            .status(401)

            .json({error: `You must be logged in to access this resource`})

    }

    next()

}
```

2. Now we will import the function from `auth.middleware.js` and replace the logic with middleware
```js
// importing enusure authenticated middelware
import {ensureAuthenticated} from '../middleware/auth.middleware.js'
router.post('/shorten', ensureAuthenticated ,async function (req, res) {}
```
