
Now we will create an authenticated POST route on `/shorten` which will shorten the long URL

1. Create a new route file under `/routes` directory named `url.routes.js`

url.routes.js
```js
// Importing express

import express from 'express'

  

// creating router from express

const router = express.Router()

  

// Creating new post route

router.post('/shorten', async function (req, res) {

    // Making sure user is authenticated

    const userID = req.user?.id

  

    // Returning response if user id does not exists

    if (!userID) return res.status(401).json({error: `You must be logged in to access this resource`})

})

  
  

// Exporting router

export default router
```

2. Creating a new schema for shorten post request under `request.validation.js` where it contains url to shorten and code (optional)
```js
export const loginPostRequestBodySchema = z.object({
    url: z.string().url(),
    code: z.string().optional()
})
```

3. Now you have to import the schema in `url.routes.js` file and perform error verification process as done in previous routes

url.routes.js
```js
// Validating result
const validationResult = await shortenPostRequestBodySchema.safeParseAsync(req.body)

// Checking and giving an returning if any error occurs
if (validationResult.error) {
return res.status(400).json({error: validationResult.error.message.format()})}

// Getting and storing url and code (optional) from url variable

```

4. Now we will create a new `urlsTable` in new model `url.model.js` file under `/models` folder using the drizzle

url.model.js
```js
// Importing required modules
import { pgTable, uuid, varchar, text, timestamp } from 'drizzle-orm/pg-core'

// Importing users table
import { usersTable } from './user.model.js'

export const urlsTable = pgTable('urls', {

    // Creating new id
    id: uuid().primaryKey().defaultRandom(),

  
    // Getting shortcode and url
    shortCode: varchar('code', {length: 155 }).notNull().unique(),
    targetURL: text('target_url').notNull(),

  
    // Getting userid based on user present in userstable
    userId: uuid('user_id').references(() => usersTable.id).notNull(),


    // Creating created and updated at using timestamp
    createdAt: timestamp('created_at').defaultNow().notNull(),
    updatedAt: timestamp('updated_at').$onUpdate(() => new Date()),
})
```

5. Now we will export the schema under `/models/index.js`
```js
export {urlsTable} from '../models/url.model.js'
```

6. Now we will create insert logic, now we have to generate code if not provided by the user using the `nanoid` , we will install it using `pnpm i nanoid` and write the final version of the url route
url.routes.js
```js
// Importing express

import express from 'express'

// importing url validation

import {shortenPostRequestBodySchema} from '../validation/request.validation.js'

// Importing db to insert url

import {db} from '../db/index.js'

// inserting models (url table)

import {urlsTable} from '../models/index.js'

// Importing nanoid

import {nanoid} from 'nanoid'

  

// creating router from express

const router = express.Router()

  

// Creating new post route

router.post('/shorten', async function (req, res) {

    // Making sure user is authenticated

    const userID = req.user.id

  

    // Returning response if user id does not exists

    if (!userID) return res.status(401).json({error: `You must be logged in to access this resource`})

  

    // Validating result

    const validationResult = await shortenPostRequestBodySchema.safeParseAsync(req.body)

  

    // Checking and giving an returning if any error occurs

    if (validationResult.error) {

        return res.status(400).json({error: validationResult.error.message})

    }

    // Getting and storing url from url variable

    const {url, code} = validationResult.data

  

    // Inserting the code in table if provided else generated by nanoid of 6 chars

    const shortCode = code ?? nanoid(6);

  

    // Inserting url and code (if given) and storing it in result

    const [result] = await db.insert(urlsTable).values({

        // Getting short code

        shortCode,

        // targetUrl

        targetURL: url,

        // user id

        userId: req.user.id

    }).returning({

        id: urlsTable.id,

        shortCode: urlsTable.shortCode,

        targetURL: urlsTable.targetURL

    })

  

    // Retruning id, shortcode and target url

    return res.status(201).json({

        id: result.id,

        shortCode: result.shortCode,

        targetURL: result.targetURL

    })

})

  
  

// Exporting router

export default router
```

7. Registering the route in main `index.js`

index.js
```js
// Importing url Router
import urlRouter from './routes/url.routes.js'

app.use(urlRouter)
```
