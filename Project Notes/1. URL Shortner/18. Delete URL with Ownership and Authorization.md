
1. We will make sure that it only deletes url short-code which belongs to the login user by comparing 2 conditions using `and` from drizzle orm

url.routes.js
```js
// Creating a DELETE route to delete short code by code id

router.delete('/:id', ensureAuthenticated, async function (req, res) {

    // Getting id from url parameter

    const id = req.params.id

  

    // Deleting only user assigned codes

    await db.delete(urlsTable)

        .where(and(eq(urlsTable.id, id), eq(urlsTable.userId, req.user.id)))

})
```
